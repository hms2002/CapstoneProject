    public static void ApplyInPlaceNavigation()
    {
        // Only the settings root gains an authored visibility group. Keep every
        // other existing block, including localized text and geometry, unchanged.
        string original = File.ReadAllText(GlobalPath);
        var persistent = AssetDatabase.LoadAssetAtPath<GameObject>(GlobalPath).GetComponentInChildren<SettingsPanelUI>(true);
        var ids = VisibilityIds(persistent);
        EditPrefab(GlobalPath, root => AuthorVisibility(root.GetComponentInChildren<SettingsPanelUI>(true)), "PopupCanvas/SettingUI");
        PreserveOutsideScope(GlobalPath, original, ids);
        original = File.ReadAllText(TitlePath);
        var scene = EditorSceneManager.OpenScene(TitlePath, OpenSceneMode.Single);
        var panel = scene.GetRootGameObjects().SelectMany(r => r.GetComponentsInChildren<SettingsPanelUI>(true)).Single();
        ids = VisibilityIds(panel);
        AuthorVisibility(panel);
        EditorSceneManager.SaveScene(scene);
        PreserveOutsideScope(TitlePath, original, ids);
        Unity.CodeEditor.CodeEditor.CurrentEditor.SyncAll();
        Debug.Log("SETTINGS_IN_PLACE_AUTHOR_PASS");
    }

    static HashSet<ulong> VisibilityIds(SettingsPanelUI panel) => panel.GetComponents<Component>().Cast<Object>()
        .Append(panel.gameObject).Select(o => GlobalObjectId.GetGlobalObjectIdSlow(o).targetObjectId).ToHashSet();

    static void AuthorVisibility(SettingsPanelUI panel)
    {
        var group = panel.GetComponent<CanvasGroup>();
        if (group == null) group = panel.gameObject.AddComponent<CanvasGroup>();
        group.alpha = 1; group.interactable = true; group.blocksRaycasts = true;
        var serialized = new SerializedObject(panel);
        serialized.FindProperty("temporaryHiddenCanvasGroup").objectReferenceValue = group;
        serialized.ApplyModifiedPropertiesWithoutUndo();
    }

    [MenuItem("Tools/UI/Apply Approved Settings Skin")]
    public static void Apply()
    {
        ImportArt();
        AddTranslations();
        EditPrefab(RowPath, StyleKeyRow, "");
        EditPrefab(GlobalPath, StylePanels, "PopupCanvas/SettingUI", "PopupCanvas/KeyMappingUI");
        string original = File.ReadAllText(TitlePath);
        var scene = EditorSceneManager.OpenScene(TitlePath, OpenSceneMode.Single);
        var panels = scene.GetRootGameObjects().SelectMany(r => r.GetComponentsInChildren<SettingsPanelUI>(true)
            .Select(p => p.gameObject).Concat(r.GetComponentsInChildren<KeyBindingPanelUI>(true).Select(p => p.gameObject))).ToArray();
        var ids = CaptureIds(panels);
        textStyles.Clear();
        foreach (var panel in panels)
            if (panel.TryGetComponent<SettingsPanelUI>(out var settings)) StyleSettings(settings); else StyleKeys(panel.GetComponent<KeyBindingPanelUI>());
        foreach (Action style in textStyles) style();
        EditorSceneManager.SaveScene(scene);
        PreserveOutsideScope(TitlePath, original, ids);
        Unity.CodeEditor.CodeEditor.CurrentEditor.SyncAll();
        Debug.Log("SETTINGS_SKIN_AUTHOR_PASS");
    }

    static void ImportArt()
    {
        string path = ArtPath + "SettingsSkin.png";
        AssetDatabase.ImportAsset(path, ImportAssetOptions.ForceSynchronousImport);
        var importer = (TextureImporter)AssetImporter.GetAtPath(path);
        importer.textureType = TextureImporterType.Sprite; importer.spriteImportMode = SpriteImportMode.Multiple;
        importer.spritePixelsPerUnit = 100; importer.filterMode = FilterMode.Point;
        importer.mipmapEnabled = false; importer.textureCompression = TextureImporterCompression.Uncompressed;
        importer.alphaIsTransparency = true; importer.npotScale = TextureImporterNPOTScale.None;
        importer.wrapMode = TextureWrapMode.Clamp;
        var settings = new TextureImporterSettings(); importer.ReadTextureSettings(settings);
        settings.spriteMeshType = SpriteMeshType.FullRect; importer.SetTextureSettings(settings);
        SpriteMetaData Slice(string name, int x, int y, int w, int h, int border) => new()
        { name = name, rect = new Rect(x, 64-y-h, w, h), pivot = Vector2.one * .5f,
            alignment = (int)SpriteAlignment.Center, border = Vector4.one * border };
        importer.spritesheet = new[] { Slice("Panel",0,0,32,32,5), Slice("Row",40,0,32,32,5),
            Slice("Button",80,0,32,32,5), Slice("Key",120,0,32,32,5), Slice("Handle",160,0,16,24,3),
            Slice("Track",184,0,16,12,1), Slice("Fill",208,0,16,12,1), Slice("Divider",232,0,8,2,0), Slice("Reset",232,16,12,12,0) };
        importer.SaveAndReimport();
        foreach (var sprite in AssetDatabase.LoadAllAssetsAtPath(path).OfType<Sprite>()) sprites[sprite.name] = sprite;
    }

    static void EditPrefab(string path, Action<GameObject> edit, params string[] approved)
    {
        string original = File.ReadAllText(path);
        var persistent = AssetDatabase.LoadAssetAtPath<GameObject>(path);
        var ids = CaptureIds(approved.Select(p => string.IsNullOrEmpty(p) ? persistent : persistent.transform.Find(p).gameObject));
        var preservation = UiPrefabPresentationPreservation.Capture(persistent, approved);
        var root = PrefabUtility.LoadPrefabContents(path);
        textStyles.Clear();
        try
        {
            preservation.BindLoadedRoot(root); edit(root); preservation.RestoreBeforeSave();
            foreach (Action style in textStyles) style();
            PrefabUtility.SaveAsPrefabAsset(root, path);
        }
        finally { PrefabUtility.UnloadPrefabContents(root); textStyles.Clear(); }
        PreserveOutsideScope(path, original, ids);
    }

    static HashSet<ulong> CaptureIds(IEnumerable<GameObject> roots)
    {
        var ids = new HashSet<ulong>();
        foreach (var root in roots)
            foreach (var t in root.GetComponentsInChildren<Transform>(true))
                foreach (var o in t.GetComponents<Component>().Cast<Object>().Append(t.gameObject))
                    if (o != null) ids.Add(GlobalObjectId.GetGlobalObjectIdSlow(o).targetObjectId);
        return ids;
    }

    static void PreserveOutsideScope(string path, string original, HashSet<ulong> approved)
    {
        string[] Blocks(string text) => Regex.Split(text, @"(?=^--- !u!)", RegexOptions.Multiline);
        ulong Id(string block) => ulong.TryParse(Regex.Match(block, @"^--- !u!\d+ &(\d+)").Groups[1].Value, out var id) ? id : 0;
        var old = Blocks(original).Where(b => Id(b) != 0).ToDictionary(Id);
        var current = Blocks(File.ReadAllText(path));
        for (int i = 0; i < current.Length; i++)
        {
            ulong id = Id(current[i]);
            if (!old.TryGetValue(id, out var before)) continue;
            if (!approved.Contains(id)) { current[i] = before; continue; }
            // Native localization callbacks must not migrate existing authored fonts/materials.
            foreach (string field in new[] { "m_fontAsset", "m_sharedMaterial", "m_fontSharedMaterials" })
            {
                string pattern = @"^  " + field + @":.*(?:\r?\n  -.*)*";
                var saved = Regex.Match(before, pattern, RegexOptions.Multiline);
                if (saved.Success) current[i] = Regex.Replace(current[i], pattern, _ => saved.Value, RegexOptions.Multiline);
            }
        }
        var remaining = current.Select(Id).ToHashSet();
        var missing=old.Keys.Where(id=>!remaining.Contains(id)).ToArray();
        if(missing.Any(id=>!old[id].Split('\n')[0].Contains(" stripped")))
            throw new InvalidOperationException("Authoring removed an existing object in " + path);
        // Native saves can prune unrelated stripped references before the preserved
        // PrefabInstance overrides are restored. Retain their original targets too.
        // Asset readers can memory-map the current file on Windows, preventing
        // truncation (Win32 1224). Replace it atomically after writing our copy.
        string staged = path + ".codex-writing";
        try
        {
            File.WriteAllText(staged, string.Concat(current)+string.Concat(missing.Select(id=>old[id])), new UTF8Encoding(false));
            File.Replace(staged, path, null);
        }
        finally { if (File.Exists(staged)) File.Delete(staged); }
        AssetDatabase.ImportAsset(path, ImportAssetOptions.ForceSynchronousImport);
    }

    {
        var bg = (RectTransform)panel.transform.Find("Background");

        foreach (var layout in bg.GetComponentsInChildren<LayoutGroup>(true)) layout.enabled = false;

        bg.GetComponent<ScrollRect>().enabled = false;
        bg.Find("Scrollbar").gameObject.SetActive(false);
        foreach (Transform child in bg)
            if (child.name.Contains("ButtonChain")) child.gameObject.SetActive(false);
        var mask = (RectTransform)bg.Find("ContentsMask");
        foreach (var b in mask.GetComponents<Behaviour>()) b.enabled = false;
        Stretch(mask, 28, 120, 28, 62);
        Stretch((RectTransform)mask.Find("Contents"));
        var title = (RectTransform)bg.Find("TitleTextPanel");
        Fixed(title,.5f,1,0,0,320,76); Skin(title.GetComponent<Image>(),"Panel");
        var titleText = title.GetComponentInChildren<TMP_Text>(true);
        Stretch(titleText.rectTransform,12,8,12,8); TextStyle(titleText,34,"8cdbec",TextAlignmentOptions.Center);
        // These anchors are the existing endpoint/constraint references.
        Fixed((RectTransform)bg.Find("BottomAnchor"),.18f,1,0,-4,0,0);
        Fixed((RectTransform)bg.Find("BottomAnchor (1)"),.82f,1,0,-4,0,0);
        var chains = panel.GetComponentsInChildren<SettingsPanelFakeChainPresentation>(true)
            .Where(c => c.transform.parent == panel.transform).ToArray();
        for (int i=0;i<chains.Length;i++)
        {
            var so = new SerializedObject(chains[i]);
            var top = (RectTransform)so.FindProperty("topAnchor").objectReferenceValue;
            var bottom = (RectTransform)so.FindProperty("bottomAnchor").objectReferenceValue;
            Vector3 endpoint = bottom.position;
            top.position = endpoint + panel.transform.TransformVector(new Vector3(0,600,0));
            chains[i].CaptureAuthoredSegmentLengths();
        }
        return bg;
    }

    static void Footer(RectTransform bg,TMP_Text sample,bool keys,Component panel)
    {
        var rule=GetImage(Ensure(bg,"FooterDivider"));BottomRect(rule.rectTransform,28,110,-56,2,true);Skin(rule,"Divider");
        var hint=Label(bg,"FooterHint",sample,keys?"keys_hint":"settings_hint");BottomRect(hint.rectTransform,28,76,-56,30,true);TextStyle(hint,20,"b9afce",TextAlignmentOptions.Left);
        var group=(RectTransform)bg.Find("ButtonGroup");BottomRect(group,28,18,-56,52,true);
        var close=Ref<Button>(panel,"closeButton");var apply=Ref<Button>(panel,keys?"applyButton":"applyDisplayButton");
        Fixed((RectTransform)close.transform.parent,0,.5f,130,0,260,52);Fixed((RectTransform)apply.transform.parent,1,.5f,-150,0,300,52);
        foreach(var button in group.GetComponentsInChildren<Button>(true))
        {
            var drop=button.GetComponent<UIChainDropPresentation>();
            if(drop!=null)
            {
                drop.enabled=false;
                // Disabled behaviours still receive Awake. Retired constraints must
                // not reposition the new fixed footer buttons during that callback.
                var state=new SerializedObject(drop);
                state.FindProperty("constrainPanelByChainReach").boolValue=false;
                state.FindProperty("snapClosedOnAwake").boolValue=false;
                state.FindProperty("playOnEnable").boolValue=false;
                state.FindProperty("chainConstraints").arraySize=0;
                state.FindProperty("fakeChainPresentation").objectReferenceValue=null;
                state.ApplyModifiedPropertiesWithoutUndo();
            }
            var cg=button.GetComponent<CanvasGroup>();if(cg!=null){cg.alpha=1;cg.interactable=true;cg.blocksRaycasts=true;}
            Stretch((RectTransform)button.transform);StyleButton(button,26);
        }
        if(keys){var reset=Ref<Button>(panel,"resetAllButton");Fixed((RectTransform)reset.transform.parent,1,.5f,-460,0,280,52);}
    }

    static TMP_Text Label(Transform parent,string name,TMP_Text sample,string key)
    {
        var existing=parent.Find(name);TMP_Text label;
        if(existing!=null)label=existing.GetComponent<TMP_Text>();
        else{label=Object.Instantiate(sample,parent);label.name=name;}
        var localize=label.GetComponent<LocalizeStringEvent>() ?? label.gameObject.AddComponent<LocalizeStringEvent>();
        localize.StringReference=new LocalizedString(GameText.TableName,"ui.settings_skin."+key);
        if(localize.OnUpdateString.GetPersistentEventCount()==0)
            UnityEventTools.AddPersistentListener(localize.OnUpdateString,(UnityEngine.Events.UnityAction<string>)Delegate.CreateDelegate(typeof(UnityEngine.Events.UnityAction<string>),label,typeof(TMP_Text).GetProperty("text").GetSetMethod()));
        label.text=Translations.First(r=>r[0]==key)[1];
        return label;
    }

    static void InspectRoot(GameObject root, StringBuilder report)
    {
        foreach (var panel in root.GetComponentsInChildren<SettingsPanelUI>(true))
        {
            report.AppendLine("SETTINGS " + root.name + "/" + PathOf(root.transform, panel.transform));
            References(panel, report); Dump(panel.transform, "", report);
        }
        foreach (var panel in root.GetComponentsInChildren<KeyBindingPanelUI>(true))
        {
            report.AppendLine("KEYS " + root.name + "/" + PathOf(root.transform, panel.transform));
            References(panel, report); Dump(panel.transform, "", report);
        }
    }

    static void References(Component component, StringBuilder report)
    {
        var iterator = new SerializedObject(component).GetIterator();
        while (iterator.NextVisible(true))
            if (iterator.propertyType == SerializedPropertyType.ObjectReference && iterator.objectReferenceValue is Component target)
                report.AppendLine("  REF " + iterator.propertyPath + " => " + PathOf(component.transform, target.transform));
    }

    static void Dump(Transform transform, string indent, StringBuilder report)
    {
        if (transform.name.StartsWith("ChainLink", StringComparison.Ordinal)) return;
        string components = string.Join(",", transform.GetComponents<Component>().Select(c => c == null ? "MISSING" : c.GetType().Name));
        string rect = transform is RectTransform r ? $" a={r.anchorMin}/{r.anchorMax} p={r.pivot} pos={r.anchoredPosition} size={r.sizeDelta} scale={r.localScale}" : "";
        string image = transform.TryGetComponent<UnityEngine.UI.Image>(out var im) ? $" sprite={(im.sprite != null ? im.sprite.name : "none")} type={im.type} color={im.color}" : "";
        string text = transform.TryGetComponent<TMP_Text>(out var tmp) ? $" text={tmp.text.Replace("\n", "|")} font={(tmp.font != null ? tmp.font.name : "none")} fs={tmp.fontSize}" : "";
        report.AppendLine(indent + transform.name + " active=" + transform.gameObject.activeSelf + " [" + components + "]" + rect + image + text);
        foreach (Transform child in transform) Dump(child, indent + "  ", report);
    }

    static string PathOf(Transform root, Transform target)
    {
        string result = target.name;
        while (target.parent != null && target != root)
        { target = target.parent; if (target != root) result = target.name + "/" + result; }
        return result;
    }

    [MenuItem("Tools/Encyclopedia/Apply Approved Pixel Skin And Discovery")]
    public static void Apply()
    {
        ImportArt();
        string path = EncyclopediaCreatureUIAuthoring.PrefabPath;
        string before = File.ReadAllText(path);
        var source = AssetDatabase.LoadAssetAtPath<GameObject>(path);
        var tab = source.GetComponentInChildren<EncyclopediaCreatureTab>(true);
        var detail = source.GetComponentInChildren<EncyclopediaCreatureDetailView>(true);
        var ts = new SerializedObject(tab); var ds = new SerializedObject(detail);
        var approvedRoots = new[] { Field<GameObject>(ts,"leftContentRoot"), Field<GameObject>(ts,"rightContentRoot"), Field<GameObject>(ds,"fullViewRoot") };
        var allowed = new HashSet<ulong>();
        foreach (var approved in approvedRoots)
            foreach (Transform node in approved.GetComponentsInChildren<Transform>(true))
                foreach (Object obj in node.GetComponents<Component>().Cast<Object>().Append(node.gameObject))
                    if (obj != null) allowed.Add(GlobalObjectId.GetGlobalObjectIdSlow(obj).targetObjectId);
        allowed.Add(GlobalObjectId.GetGlobalObjectIdSlow(tab).targetObjectId);
        var preservation = UiPrefabPresentationPreservation.Capture(source,
            approvedRoots.Select(r => AnimationUtility.CalculateTransformPath(r.transform, source.transform)).ToArray());
        var root = PrefabUtility.LoadPrefabContents(path);
        try
        {
            preservation.BindLoadedRoot(root);
            Style(root);
            preservation.RestoreBeforeSave();
            var styledDetail = new SerializedObject(root.GetComponentInChildren<EncyclopediaCreatureDetailView>(true));
            var title = Field<TMP_Text>(styledDetail, "characterTitleText");
            title.fontSize = 24;
            title.enableAutoSizing = false;
            PrefabUtility.SaveAsPrefabAsset(root, path);
        }
        finally { PrefabUtility.UnloadPrefabContents(root); }
        Preserve(path, before, allowed);
        Unity.CodeEditor.CodeEditor.CurrentEditor.SyncAll();
        Directory.CreateDirectory(Evidence);
        File.WriteAllLines(Evidence+"authoring-allowed-ids.txt",allowed.OrderBy(id=>id).Select(id=>id.ToString()));
        Debug.Log("ENCYCLOPEDIA_SKIN_AUTHOR_PASS");
    }

#pragma warning disable CS0618
    private static void ImportArt()
    {
        sprites.Clear();
        string path = Art + "EncyclopediaPixelSkin.png";
        var manifest = JsonUtility.FromJson<Manifest>(File.ReadAllText(Art+"slices.json"));
        AssetDatabase.ImportAsset(path, ImportAssetOptions.ForceSynchronousImport);
        var importer = (TextureImporter)AssetImporter.GetAtPath(path);
        Configure(importer);
        importer.spriteImportMode = SpriteImportMode.Multiple;
        importer.spritesheet = manifest.sprites.Select(s => new SpriteMetaData
        {
            name=s.name, rect=new Rect(s.x,144-s.y-s.height,s.width,s.height),
            pivot=Vector2.one*.5f, alignment=(int)SpriteAlignment.Center, border=Vector4.one*s.border
        }).ToArray();
        importer.SaveAndReimport();
        foreach (Sprite sprite in AssetDatabase.LoadAllAssetsAtPath(path).OfType<Sprite>()) sprites[sprite.name]=sprite;
        foreach (string name in new[] {"BlackFadeTop","BlackFadeBottom","DotEmptyLight","DotSelectedLight"})
        {
            string overlay=Art+name+".png";
            AssetDatabase.ImportAsset(overlay,ImportAssetOptions.ForceSynchronousImport);
            var single=(TextureImporter)AssetImporter.GetAtPath(overlay);
            Configure(single);single.spriteImportMode=SpriteImportMode.Single;single.SaveAndReimport();
            sprites[name]=AssetDatabase.LoadAssetAtPath<Sprite>(overlay);
        }
        if (sprites.Count!=27) throw new InvalidOperationException("Expected 23 skin slices and four overlays.");
    }
#pragma warning restore CS0618
    private static void Configure(TextureImporter importer)
    {
        importer.textureType=TextureImporterType.Sprite;importer.spritePixelsPerUnit=100;
        importer.filterMode=FilterMode.Point;importer.mipmapEnabled=false;importer.alphaIsTransparency=true;
        importer.textureCompression=TextureImporterCompression.Uncompressed;
        importer.npotScale=TextureImporterNPOTScale.None;importer.wrapMode=TextureWrapMode.Clamp;
        var settings=new TextureImporterSettings();importer.ReadTextureSettings(settings);
        settings.spriteMeshType=SpriteMeshType.FullRect;importer.SetTextureSettings(settings);
    }

    private static void Frame(Transform parent,string sprite)
    {
        var frame=parent.Find("Frame");
        if(frame==null) frame=Decoration("Frame",parent).transform;
        foreach(Transform child in frame) child.gameObject.SetActive(false);
        var image=frame.GetComponent<Image>() ?? frame.gameObject.AddComponent<Image>();
        Stretch(image.rectTransform);Skin(image,sprite);frame.SetAsLastSibling();
    }
    private static void Fade(Transform node,string sprite)
    {
        var gradient=node.GetComponent<EncyclopediaGradientGraphic>();if(gradient!=null) gradient.enabled=false;
        Skin(Decoration("PixelFade",node),sprite,false);
    }

        button.spriteState=new UnityEngine.UI.SpriteState{highlightedSprite=sprites["ButtonHover"],pressedSprite=sprites["ButtonPressed"],selectedSprite=sprites["ButtonHover"],disabledSprite=sprites["ButtonNormal"]};
        foreach(var text in button.GetComponentsInChildren<TMP_Text>(true)) text.gameObject.SetActive(false);
        var image=Decoration("PixelIcon",button.transform);Skin(image,icon,false);image.preserveAspect=true;
        image.rectTransform.anchorMin=image.rectTransform.anchorMax=Vector2.one*.5f;
        image.rectTransform.sizeDelta=new Vector2(icon=="FullViewExpand"?40:30,icon=="FullViewExpand"?40:30);image.rectTransform.anchoredPosition=Vector2.zero;
    }
    private static Image Decoration(string name,Transform parent)
    {
        var found=parent.Find(name);if(found!=null) return found.GetComponent<Image>();
        var go=new GameObject(name,typeof(RectTransform),typeof(UnityEngine.CanvasRenderer),typeof(Image));go.transform.SetParent(parent,false);
        var image=go.GetComponent<Image>();Stretch(image.rectTransform);image.raycastTarget=false;return image;
    }
    private static void Stretch(RectTransform rect) {rect.anchorMin=Vector2.zero;rect.anchorMax=Vector2.one;rect.offsetMin=rect.offsetMax=Vector2.zero;}
    private static void Skin(Image image,string sprite,bool sliced=true)
    {image.sprite=sprites[sprite];image.color=Color.white;image.type=sliced?Image.Type.Sliced:Image.Type.Simple;image.pixelsPerUnitMultiplier=.4f;image.raycastTarget=image.GetComponent<Button>()!=null;}
    private static T Field<T>(SerializedObject owner,string name) where T:Object => (T)owner.FindProperty(name).objectReferenceValue;
    private static void Ref(SerializedObject owner,string name,Object value) => owner.FindProperty(name).objectReferenceValue=value;
    private static void Refs(SerializedObject owner,string name,Object[] values)
    {var p=owner.FindProperty(name);p.arraySize=values.Length;for(int i=0;i<values.Length;i++)p.GetArrayElementAtIndex(i).objectReferenceValue=values[i];}

    private static void Preserve(string path,string original,HashSet<ulong> allowed)
    {
        string[] Blocks(string text)=>Regex.Split(text,@"(?=^--- !u!)",RegexOptions.Multiline);
        ulong Id(string block)=>ulong.TryParse(Regex.Match(block,@"^--- !u!\d+ &(\d+)").Groups[1].Value,out var id)?id:0;
        var before=Blocks(original).Where(b=>Id(b)!=0).ToDictionary(Id);var after=Blocks(File.ReadAllText(path));
        for(int i=0;i<after.Length;i++)
        {
            ulong id=Id(after[i]);if(!before.TryGetValue(id,out string old))continue;
            if(!allowed.Contains(id)){after[i]=old;continue;}
            foreach(string field in new[]{"m_fontAsset","m_sharedMaterial","m_fontSharedMaterials"})
            {string pattern=@"^  "+field+@":.*(?:\r?\n  -.*)*";var saved=Regex.Match(old,pattern,RegexOptions.Multiline);if(saved.Success)after[i]=Regex.Replace(after[i],pattern,_=>saved.Value,RegexOptions.Multiline);}
        }
        var remaining=after.Select(Id).ToHashSet();var missing=before.Keys.Where(id=>!remaining.Contains(id)).ToArray();
        if(missing.Any(id=>!before[id].Split('\n')[0].Contains(" stripped")))throw new InvalidOperationException("Existing prefab object removed.");
        string staged=path+".codex-writing";
        try {File.WriteAllText(staged,string.Concat(after)+string.Concat(missing.Select(id=>before[id])),new UTF8Encoding(false));File.Replace(staged,path,null);}
        finally {if(File.Exists(staged))File.Delete(staged);}
        AssetDatabase.ImportAsset(path,ImportAssetOptions.ForceSynchronousImport);
    }
